// App.js
import React, { useState } from "react";
import { StyleSheet, Text, View, TextInput, TouchableOpacity, ActivityIndicator } from "react-native";

export default function App() {
  const [screen, setScreen] = useState("home");
  const [form, setForm] = useState({ pickup: "", dropoff: "", notes: "" });
  const [loading, setLoading] = useState(false);
  const [booking, setBooking] = useState(null);

  const handleChange = (key, value) => {
    setForm((prev) => ({ ...prev, [key]: value }));
  };

  const requestAmbulance = () => {
    setLoading(true);
    setTimeout(() => {
      setBooking({
        bookingId: Math.floor(Math.random() * 10000),
        status: "On the way",
        eta: 5,
        ambulance: { id: "AMB-21", driver: "John Doe" },
      });
      setScreen("tracking");
      setLoading(false);
    }, 2000);
  };

  function renderBooking() {
    return (
      <View style={styles.container}>
        <Text style={styles.title}>Book Ambulance</Text>

        <TextInput
          placeholder="Pickup location"
          style={styles.input}
          value={form.pickup}
          onChangeText={(t) => handleChange("pickup", t)}
        />
        <TextInput
          placeholder="Drop-off hospital (optional)"
          style={styles.input}
          value={form.dropoff}
          onChangeText={(t) => handleChange("dropoff", t)}
        />
        <TextInput
          placeholder="Notes"
          style={[styles.input, { height: 80 }]}
          multiline
          value={form.notes}
          onChangeText={(t) => handleChange("notes", t)}
        />

        {loading ? (
          <ActivityIndicator size="large" color="#E63946" style={{ marginTop: 20 }} />
        ) : (
          <TouchableOpacity style={styles.button} onPress={requestAmbulance}>
            <Text style={styles.btnText}>Request Ambulance</Text>
          </TouchableOpacity>
        )}

        <TouchableOpacity style={styles.linkBtn} onPress={() => setScreen("home")}>
          <Text style={{ color: "#333" }}>Back</Text>
        </TouchableOpacity>
      </View>
    );
  }

  function renderTracking() {
    if (!booking) return <Text>No booking yet</Text>;

    return (
      <View style={styles.container}>
        <Text style={styles.title}>Tracking</Text>
        <Text style={styles.text}>
          <Text style={{ fontWeight: "600" }}>Booking:</Text> {booking.bookingId}
        </Text>
        <Text style={styles.text}>
          <Text style={{ fontWeight: "600" }}>Status:</Text> {booking.status}
        </Text>
        <Text style={styles.text}>
          <Text style={{ fontWeight: "600" }}>ETA:</Text> {booking.eta} min
        </Text>
        <Text style={styles.text}>
          <Text style={{ fontWeight: "600" }}>Ambulance:</Text> {booking.ambulance?.id} —{" "}
          {booking.ambulance?.driver}
        </Text>

        <TouchableOpacity style={styles.linkBtn} onPress={() => setScreen("booking")}>
          <Text style={{ color: "#333" }}>Back to Booking</Text>
        </TouchableOpacity>
      </View>
    );
  }

  return (
    <View style={styles.container}>
      {screen === "home" && (
        <View style={styles.container}>
          <Text style={styles.title}>🚑 ResQ</Text>
          <TouchableOpacity style={styles.button} onPress={() => setScreen("booking")}>
            <Text style={styles.btnText}>Book Ambulance</Text>
          </TouchableOpacity>
        </View>
      )}
      {screen === "booking" && renderBooking()}
      {screen === "tracking" && renderTracking()}
    </View>
  );
}

const styles = StyleSheet.create({
  container: { flex: 1, alignItems: "center", justifyContent: "center", padding: 20 },
  title: { fontSize: 22, fontWeight: "700", marginBottom: 20 },
  input: {
    width: "100%",
    borderWidth: 1,
    borderColor: "#ccc",
    borderRadius: 10,
    padding: 10,
    marginBottom: 15,
  },
  button: {
    backgroundColor: "#E63946",
    padding: 15,
    borderRadius: 10,
    marginTop: 10,
    width: "100%",
    alignItems: "center",
  },
  btnText: { color: "#fff", fontWeight: "600" },
  linkBtn: { marginTop: 15 },
  text: { fontSize: 16, marginVertical: 5 },
});
